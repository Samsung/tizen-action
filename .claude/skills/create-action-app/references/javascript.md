# JavaScript (Tizen Web app) action provider

Modeled on `ActionSampleAppJs`. Placeholders: app `MyActionAppJs`, package id
`7ZznEYATQu` (use the id of your signing profile's certificate), app id
`7ZznEYATQu.MyActionAppJs`, category `Tizen.Action.Browser`, stub base `ImplBrowser`,
dlog tag `TIDL_ACTION_MYACTIONAPP`.

## Layout

```
MyActionAppJs/
├── config.xml
├── index.html
├── css/style.css
├── icon.png
├── js/ImplBrowser.js   # generated — do not edit
├── js/main.js
├── gen.sh
├── build.sh
├── run.sh
├── install_rpms.sh     # optional: push/force-install TIDL runtime RPMs
├── rpms/               # optional: parcel, rpc-port, webapi-plugins, ...
└── README.md
```

The stub targets the `tizen.rpcport` WebAPI provided by **webapi-plugins** at runtime;
the device needs a build of it that includes the rpc-port plugin. Include the RPMs plus
`install_rpms.sh` only when the runtime has to be upgraded on the target.

## config.xml

```xml
<?xml version="1.0" encoding="UTF-8"?>
<widget xmlns="http://www.w3.org/ns/widgets" xmlns:tizen="http://tizen.org/ns/widgets" version="1.0.0" viewmodes="maximized">
  <tizen:application id="7ZznEYATQu.MyActionAppJs" package="7ZznEYATQu" required_version="9.0"/>
  <feature name="http://tizen.org/feature/screen.size.all"/>
  <icon src="icon.png"/>
  <content src="index.html"/>
  <name>MyActionAppJs</name>
  <!-- Keep the running page (and the listening rpc-port stub) on relaunch:
       a proxy connecting triggers a default-operation app-control; reload=disable
       makes WRT fire the 'appcontrol' event instead of reloading the page. -->
  <tizen:app-control>
    <tizen:src name="index.html" reload="disable"/>
    <tizen:operation name="http://tizen.org/appcontrol/operation/default"/>
  </tizen:app-control>
  <!-- The server must keep processing rpc-port requests while in the background
       (a UI web app's JS is otherwise suspended once the client takes the foreground). -->
  <tizen:setting background-support="enable"/>
  <!-- Provider registration is per action. -->
  <tizen:metadata key="http://tizen.org/metadata/action/provider" value="Tv_Tizen.Action.Browser_GetCurrent"/>
  <tizen:metadata key="http://tizen.org/metadata/action/provider" value="Tv_Tizen.Action.Browser_Go"/>
  <tizen:metadata key="http://tizen.org/metadata/action/provider" value="Tv_Tizen.Action.Browser_ToCalendar"/>
  <tizen:metadata key="http://tizen.org/metadata/action/provider" value="Tv_Tizen.Action.Browser_ToPresentation"/>
  <tizen:privilege name="http://tizen.org/privilege/datasharing"/>
  <tizen:privilege name="http://tizen.org/privilege/appmanager.launch"/>
  <tizen:privilege name="http://tizen.org/privilege/filesystem.read"/>
  <tizen:privilege name="http://tizen.org/privilege/filesystem.write"/>
  <tizen:profile name="tizen"/>
</widget>
```

Both `background-support` and the `reload="disable"` app-control are required: without
them the listening stub is suspended or destroyed as soon as a client connects.

## index.html

Load the generated stub **before** `main.js` (the stub defines globals, not modules):

```html
<script src="js/ImplBrowser.js"></script>
<script src="js/main.js"></script>
```

The sample renders a dashboard: a status pill (`#status`), one card per action showing
its last arguments/timestamp, and a `<ul id="log">` terminal with a clear button. Keep
element ids stable since `main.js` updates them by id.

## js/main.js

```js
/* MyActionAppJs — Tizen Browser Action Provider (JavaScript Sample) */

(function () {
  'use strict';

  var stub = null;

  function $(id) { return document.getElementById(id); }

  function getTimestamp() { return new Date().toLocaleTimeString(); }

  function log(line, cls) {
    // console.error is what reaches dlog, so it is greppable with dlogutil
    console.error('[TIDL_ACTION_MYACTIONAPP] ' + line);

    var logEl = $('log');
    if (!logEl) return;
    var li = document.createElement('li');
    li.textContent = '[' + getTimestamp() + '] ' + line;
    if (cls) li.className = cls;
    logEl.appendChild(li);
    logEl.scrollTop = logEl.scrollHeight;
  }

  // Extend the autogenerated base class; handlers are named on<Action>.
  class BrowserService extends TizenActionBrowserServiceBase {
    constructor(sender, instance) { super(sender, instance); }

    onCreate() {
      log('Client connected: Sender=' + this.sender + ', Instance=' + this.instance, 'success');
    }

    onTerminate() {
      log('Client disconnected: Sender=' + this.sender + ', Instance=' + this.instance, 'sys');
    }

    // TizenEntityStatus Go(TizenEntityBrowser browser_ent)
    onGo(browser_ent) {
      log('onGo triggered with browser_ent=' + JSON.stringify(browser_ent), 'info');

      if ($('go-url')) $('go-url').textContent = browser_ent.Url;
      if ($('go-title')) $('go-title').textContent = browser_ent.Title;
      if ($('go-details')) $('go-details').textContent = browser_ent.Details;
      if ($('go-time')) $('go-time').textContent = getTimestamp();

      let status = new TizenEntityStatus();
      status.Success = true;
      status.Reason = '';
      return status;
    }

    // TizenEntityStatus GetCurrent(out TizenEntityBrowser result)
    onGetCurrent(result) {
      log('onGetCurrent triggered', 'info');

      result.Id = 'current_id';
      result.Extra = 'extra';
      result.Url = 'https://tizen.org';
      result.Title = 'Tizen Project';
      result.Details = 'Details';

      let status = new TizenEntityStatus();
      status.Success = true;
      status.Reason = '';
      return status;
    }

    // ... one handler per remaining action of the category ...
  }
```

### Method signatures

Each action's method signature (as a handler method) depends on its `.action` schema.
Handler names are `on` + PascalCase action name (`onGo`, `onGetCurrent`, etc.):

- **No input, no output**: `onActionName()` (no parameters)
- **Input entity, no output**: `onActionName(entityInstance)` (entity passed as parameter)
- **No input, output entity**: `onActionName(resultObject)` (entity passed as output reference)
- **Input and output**: `onActionName(entityInstance, resultObject)` (both passed)

To find the correct signature for your category, read the generated stub:

```bash
# After running gen.sh or actionc
grep "on[A-Z]" js/ImplBrowser.js | head -20
```

Then implement each handler in `BrowserService`, matching the exact signature. All
handlers return `TizenEntityStatus()` with `Success` and `Reason` fields set.

### Service startup and listening

```js
  function start() {
    if (typeof tizen === 'undefined') { log('tizen global object is not available!', 'error'); return; }
    if (!tizen.rpcport) { log('tizen.rpcport WebAPI is not available!', 'error'); return; }
    try {
      stub = new TizenActionBrowser(function (sender, instance) {
        return new BrowserService(sender, instance);
      });
      stub.listen();
      if ($('status')) { $('status').textContent = 'Listening'; $('status').className = 'on'; }
      log('Listening on port "TizenActionBrowser"', 'success');
    } catch (e) {
      log('Listen failed: ' + e.message, 'error');
    }
  }

  window.addEventListener('load', function () {
    document.addEventListener('tizenhwkey', function (ev) {
      if (ev.keyName === 'back') {
        try { tizen.application.getCurrentApplication().exit(); } catch (ignore) {}
      }
    });
    // Relaunch arrives as an appcontrol event thanks to reload="disable".
    window.addEventListener('appcontrol', function () {
      log('Relaunched via AppControl (kept listening)', 'sys');
    });
    start();
  });
})();
```

Output parameters arrive as pre-allocated objects to mutate — assign to their
properties, do not reassign the parameter. Handlers return a `TizenEntityStatus`.
Optionally mirror the sample's file logging via
`tizen.filesystem.resolve('wgt-private-tmp', ...)` when console logs are hard to read.

## gen.sh

```bash
#!/usr/bin/env bash
#
# Generates the JavaScript TIDL interface and stub code for this sample from default-actions using actionc.
#
#   ACTIONC            path to actionc (default: PATH or ~/.action-tools/actionc)
#   ACTIONC_TIDLC      path to tidlc  (default: ~/.action-tools/tidlc)
#   ACTIONC_DATA_DIR   actions/ + entities/ dir (default: ~/.action-tools/data)

set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
JS_DIR="${SCRIPT_DIR}/js"

export ACTIONC_TIDLC="${ACTIONC_TIDLC:-${HOME}/.action-tools/tidlc}"
export ACTIONC_DATA_DIR="${ACTIONC_DATA_DIR:-${HOME}/.action-tools/data}"

find_actionc() {
  if [[ -n "${ACTIONC:-}" ]]; then echo "${ACTIONC}"; return; fi
  if command -v actionc >/dev/null 2>&1; then command -v actionc; return; fi
  if [[ -x "${HOME}/.action-tools/actionc" ]]; then echo "${HOME}/.action-tools/actionc"; return; fi
  return 1
}

ACTIONC_BIN="$(find_actionc)" || {
  echo "error: actionc not found. Please install the Tizen Action Compiler or add it to PATH." >&2
  exit 1
}

echo "actionc: ${ACTIONC_BIN}"
mkdir -p "${JS_DIR}"

# Match the schemas to the attached device's tizen-action before generating; the
# compiler binaries stay as installed. Best effort: without a device or network the
# installed data is used as is.
find_actionc_sync() {
  if command -v actionc-sync >/dev/null 2>&1; then command -v actionc-sync; return; fi
  if [[ -x "${HOME}/.action-tools/actionc-sync" ]]; then echo "${HOME}/.action-tools/actionc-sync"; return; fi
  return 1
}
if ACTIONC_SYNC_BIN="$(find_actionc_sync)"; then
  "${ACTIONC_SYNC_BIN}" --quiet || echo "warning: actionc-sync failed; generating from the installed schema data" >&2
fi

echo "[1/1] actionc: generating JS Browser action stub -> js/ImplBrowser"
( cd "${JS_DIR}" && "${ACTIONC_BIN}" -a Tizen.Action.Browser -l JS -o ImplBrowser )

echo "Done."
echo "  generated stub:      ${JS_DIR}/ImplBrowser.js"
```

## build.sh

```bash
#!/bin/bash
#
# Build helper for the MyActionAppJs Tizen Web app.
#
#   ./build.sh            # gen stub + build-web + package wgt (default: all)
#   ./build.sh gen|build|install|run|clean

set -e

SCRIPT_DIR="$(cd "$(dirname "$0")" && pwd)"
cd "$SCRIPT_DIR"

PROFILE="${PROFILE:-MyChatProfile}"     # a `tizen security-profiles` name
APP_ID="7ZznEYATQu.MyActionAppJs"

log() { echo -e "\033[1;34m[MyActionAppJs]\033[0m $*"; }
die() { echo -e "\033[1;31m[MyActionAppJs] error:\033[0m $*" >&2; exit 1; }

gen_stub() { log "Generating stub..."; ./gen.sh; }

build_pkg() {
  log "tizen build-web"
  tizen build-web -- .
  log "tizen package -t wgt -s $PROFILE"
  tizen package -t wgt -s "$PROFILE" -- .buildResult
  local wgt
  wgt="$(ls -t .buildResult/*.wgt 2>/dev/null | head -1)"
  [ -n "$wgt" ] || die "no .wgt produced"
  log "Packaged: $wgt"
}

install_wgt() {
  local wgt
  wgt="$(ls -t .buildResult/*.wgt 2>/dev/null | head -1)"
  [ -n "$wgt" ] || die "no .wgt found; run './build.sh' first"
  log "Installing $wgt on target"
  sdb install "$wgt"
}

launch_app() { log "Launching $APP_ID"; sdb shell launch_app "$APP_ID"; }

clean() { log "Cleaning build artifacts"; rm -rf .buildResult .build .sign; }

case "${1:-all}" in
  all)     gen_stub; build_pkg ;;
  gen)     gen_stub ;;
  build)   build_pkg ;;
  install) install_wgt ;;
  run)     install_wgt; launch_app ;;
  clean)   clean ;;
  *)       die "unknown command '$1' (use: all|gen|build|install|run|clean)" ;;
esac

log "done."
```

## run.sh

```bash
#!/usr/bin/env bash
#
# Automatically builds, deploys, and executes the MyActionAppJs test app on the device.
#
# Usage:
#   ./run.sh [-u URL] [-t TITLE] [-d DETAILS]

set -euo pipefail

URL="https://tizen.org"
TITLE="Tizen OS"
DETAILS="Rpi4 Board"

while [[ $# -gt 0 ]]; do
  case "$1" in
    -u) URL="$2"; shift 2 ;;
    -t) TITLE="$2"; shift 2 ;;
    -d) DETAILS="$2"; shift 2 ;;
    -h|--help) sed -n '2,10p' "$0" | sed 's/^# \{0,1\}//'; exit 0 ;;
    *) echo "unknown option: $1" >&2; exit 2 ;;
  esac
done

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"

echo "=== [1/5] Building Tizen JS Application (MyActionAppJs) ==="
"${SCRIPT_DIR}/build.sh" all

echo "=== [2/5] Deploying WGT to Tizen Target Board ==="
"${SCRIPT_DIR}/build.sh" install

echo "=== [2.5/5] Terminating any old active service processes ==="
sdb shell "killall wrt-loader || true"
sdb shell "ps -ef | grep 7ZznEYATQu | grep -v grep | awk '{print \$2}' | xargs kill -9 2>/dev/null || true"

echo "=== [3/5] Launching JS Service Application ==="
"${SCRIPT_DIR}/build.sh" run
echo "Waiting 4 seconds for JS application to initialize and listen on RPC port..."
sleep 4

echo "=== [4/5] Executing action dynamically targeting JS Application ==="
JSON="{\"id\": 12, \"params\": { \"name\": \"Tv_Tizen.Action.Browser_Go\", \"appid\": \"7ZznEYATQu.MyActionAppJs\", \"arguments\": { \"Url\": \"${URL}\", \"Title\": \"${TITLE}\", \"Details\": \"${DETAILS}\" } } }"
sdb shell "action-tool execute '${JSON}'"

echo "=== [5/5] Fetching dlog logs for TIDL_ACTION_MYACTIONAPP ==="
sdb shell "dlogutil -d | grep TIDL_ACTION_MYACTIONAPP"
```

A Web app must be launched (and be listening) before the action is executed — hence the
launch + `sleep` before `action-tool`, unlike the native/C# providers which the framework
can start on demand.

## Critical: Verify method signatures from the generated stub

**Do NOT assume action handler signatures match the examples above.** The actual signatures
are determined by the `.action` schema for each action in the category. Always inspect
the generated stub before implementing:

```bash
# After running gen.sh or actionc
grep "prototype.on[A-Z]" js/ImplBrowser.js | head -20
# Or search for function definitions
grep "function on[A-Z]" js/ImplBrowser.js
```

Copy the exact handler signature (parameter names and count) and implement it in your
`BrowserService` class. Mismatched handler signatures will cause:

- Handler never being called (wrong name or parameter count)
- `Cannot read property 'Url' of undefined` if parameters don't match expectations
- Return value not matching `TizenEntityStatus` format

If actions are not being triggered or handlers are silently ignored, check `js/ImplXxx.js`
for the **exact** handler signatures for your category.
